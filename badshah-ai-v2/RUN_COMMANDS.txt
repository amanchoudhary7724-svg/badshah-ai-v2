Setup:
python -m venv venv
venv\Scripts\activate
pip install -r requirements.txt
copy .env.example .env

ollama pull llama3.2:1b

python main.py

uvicorn badshah_ai.api.server:app --reload

Push to GitHub:
git add .
git commit -m "Add BADSHAH-AI v2 phase 1 core"
git push origin main
